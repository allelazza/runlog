var COL=["ID","Data","Tipo","Km","Passo","Minuti","Sforzo","FC","Achille","Caviglia","Schiena","Scarpe","Note","Salvato"];
function out(o){return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);}
function foglio(n){var ss=SpreadsheetApp.getActiveSpreadsheet();return ss.getSheetByName(n)||ss.insertSheet(n);}
function doGet(e){
  var a=(e&&e.parameter&&e.parameter.action)||"ping";
  if(a==="restore"){
    var b=foglio("Backup"),n=b.getLastRow()-1;
    if(n<1) return out({status:"empty"});
    var t=b.getRange(2,1,n,1).getValues().map(function(r){return String(r[0]).substring(1);}).join("");
    return out({status:"ok",saved:String(b.getRange(1,2).getValue()),backup:t});
  }
  return out({status:"ok",msg:"RunLog endpoint attivo",v:10});
}
function doPost(e){
  try{
    var d=JSON.parse(e.postData.contents);
    if(d.action==="test"){foglio("Backup").getRange(1,3).setValue("Test "+new Date());return out({status:"ok",msg:"scrittura ok"});}
    if(d.action==="backup"){
      var b=foglio("Backup");b.clearContents();
      b.getRange(1,1,1,2).setValues([["Backup RunLog (non modificare)",new Date()]]);
      var righe=[];
      for(var i=0;i<d.data.length;i+=40000) righe.push(["~"+d.data.substring(i,i+40000)]);
      b.getRange(2,1,righe.length,1).setNumberFormat("@").setValues(righe);
      return out({status:"ok",parti:righe.length});
    }
    var sh=foglio("Corse");
    if(sh.getLastRow()===0){sh.appendRow(COL);sh.getRange(1,1,1,COL.length).setFontWeight("bold");sh.setFrozenRows(1);}
    var id=d.action==="delete"?d.id:d.log[0],riga=0,n=sh.getLastRow()-1;
    if(n>0){var ids=sh.getRange(2,1,n,1).getValues();for(var j=0;j<ids.length;j++){if(String(ids[j][0])===String(id)){riga=j+2;break;}}}
    if(d.action==="delete"){if(riga) sh.deleteRow(riga);return out({status:"ok"});}
    if(riga) sh.getRange(riga,1,1,d.log.length).setValues([d.log]); else sh.appendRow(d.log);
    return out({status:"ok"});
  }catch(err){return out({status:"err",msg:String(err)});}
}
